cmake_minimum_required(VERSION 3.16)
project(DispWeave VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(cmake/Dependencies.cmake)
add_library(invlm_physics STATIC third_party/invLM/src/dispersion.cpp
  third_party/invLM/src/secfunc.cpp third_party/invLM/src/secfunc_fast_scalar.cpp
  third_party/invLM/src/toms748.cpp third_party/invLM/src/dispersion_kernel.cpp
  third_party/invLM/src/dunkin_derivative.cpp)
target_include_directories(invlm_physics PUBLIC third_party/invLM/include)
target_link_libraries(invlm_physics PUBLIC dw_eigen)
add_library(dispweave_core src/model.cpp src/forward.cpp src/sensitivity.cpp src/parameterization.cpp src/objective.cpp src/lm.cpp src/resolution.cpp src/selection.cpp)
target_include_directories(dispweave_core PUBLIC include)
target_link_libraries(dispweave_core PUBLIC dw_eigen PRIVATE invlm_physics)
add_executable(dispweave src/main.cpp src/io.cpp src/workflow.cpp src/resolution_io.cpp)
target_link_libraries(dispweave PRIVATE dispweave_core dw_toml dw_cli)
configure_file(cmake/version.hpp.in generated/version.hpp @ONLY)
target_include_directories(dispweave PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
foreach(target dispweave_core dispweave)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_executable(objective_example examples/objective/evaluate.cpp)
target_link_libraries(objective_example PRIVATE dispweave_core)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME selection_workflow COMMAND ${Python3_EXECUTABLE}
      ${PROJECT_SOURCE_DIR}/tests/selection_workflow.py $<TARGET_FILE:dispweave> ${PROJECT_SOURCE_DIR})
    add_test(NAME resolution_workflow COMMAND ${Python3_EXECUTABLE}
      ${PROJECT_SOURCE_DIR}/tests/resolution_workflow.py $<TARGET_FILE:dispweave> ${PROJECT_SOURCE_DIR})
    add_test(NAME inversion_workflow COMMAND ${Python3_EXECUTABLE}
      ${PROJECT_SOURCE_DIR}/tests/workflow_tests.py $<TARGET_FILE:dispweave> ${PROJECT_SOURCE_DIR})
  endif()
  add_executable(selection_physics tests/selection_physics.cpp)
  target_link_libraries(selection_physics PRIVATE dispweave_core)
  add_test(NAME selection_physics COMMAND selection_physics)
  add_executable(selection_tests tests/selection_tests.cpp)
  target_link_libraries(selection_tests PRIVATE dispweave_core)
  add_test(NAME selection COMMAND selection_tests)
  add_executable(resolution_tests tests/resolution_tests.cpp)
  target_link_libraries(resolution_tests PRIVATE dispweave_core)
  foreach(case analytic invariance validation)
    add_test(NAME resolution_${case} COMMAND resolution_tests ${case})
  endforeach()
  add_executable(lm_tests tests/lm_tests.cpp)
  target_link_libraries(lm_tests PRIVATE dispweave_core)
  foreach(case linear nonlinear failures)
    add_test(NAME lm_${case} COMMAND lm_tests ${case})
  endforeach()
  add_test(NAME objective_example COMMAND objective_example)
  add_executable(forward_tests tests/forward_tests.cpp)
  target_link_libraries(forward_tests PRIVATE dispweave_core)
  foreach(case validation halfspace golden independent scenarios ordering subdivision cutoff numerical)
    add_test(NAME ${case} COMMAND forward_tests ${case})
  endforeach()
  add_executable(sensitivity_tests tests/sensitivity_tests.cpp)
  target_link_libraries(sensitivity_tests PRIVATE dispweave_core)
  foreach(case derivatives diagnostics halfspace)
    add_test(NAME sensitivity_${case} COMMAND sensitivity_tests ${case})
  endforeach()
  add_executable(objective_tests tests/objective_tests.cpp)
  target_link_libraries(objective_tests PRIVATE dispweave_core)
  foreach(case covariance parameterization posterior objective_failures)
    add_test(NAME ${case} COMMAND objective_tests ${case})
  endforeach()
  add_test(NAME cli COMMAND ${CMAKE_COMMAND}
    -DEXE=$<TARGET_FILE:dispweave> -DSOURCE=${PROJECT_SOURCE_DIR}
    -DWORK=${PROJECT_BINARY_DIR}/cli-test -P ${PROJECT_SOURCE_DIR}/tests/cli.cmake)
endif()
add_executable(sensitivity_benchmark benchmarks/sensitivity_benchmark.cpp)
target_link_libraries(sensitivity_benchmark PRIVATE dispweave_core)
add_executable(forward_benchmark benchmarks/forward_benchmark.cpp)
target_link_libraries(forward_benchmark PRIVATE dispweave_core)
include(GNUInstallDirs)
install(TARGETS dispweave RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/dispweave)
foreach(dependency eigen toml11 CLI11 invLM)
  file(GLOB notices "${PROJECT_SOURCE_DIR}/third_party/${dependency}/LICENSE"
    "${PROJECT_SOURCE_DIR}/third_party/${dependency}/COPYING*"
    "${PROJECT_SOURCE_DIR}/third_party/${dependency}/provenance.json"
    "${PROJECT_SOURCE_DIR}/third_party/${dependency}/PATCHES.md")
  install(FILES ${notices} DESTINATION ${CMAKE_INSTALL_DATADIR}/dispweave/third_party/${dependency})
endforeach()
